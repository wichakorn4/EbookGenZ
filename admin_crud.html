ℹ️ การทำงานของระบบ SQL Query Editor ในหน้านี้:

                    
    
                        
  • โหมดปกติ (ไม่ต้องติดตั้งอะไรเพิ่ม): คุณสามารถรันคำสั่ง SELECT, INSERT, UPDATE, DELETE บนฐานข้อมูล Supabase ได้ทันที! ระบบมี Smart PostgREST Translator แปลงคำสั่ง SQL ให้โดยอัตโนมัติ
    • 
                        
  • โหมด Full Raw SQL: หากต้องการใช้ฟังก์ชัน SQL ขั้นสูง เช่น JOIN ข้ามตาราง, GROUP BY, คำนวณยอด Aggregate หรือคำสั่ง DDL สร้างตาราง ให้ทำตาม 2 ขั้นตอนด้านล่างนี้ครับ
    •

                        ไปที่ Supabase Dashboard -> เมนู SQL Editor
                    

                    
เปิดหน้าต่าง Supabase ของคุณ เลือกเมนู SQL Editor แถบด้านซ้าย แล้วกดปุ่ม + New query

                


                

                    

                        2
                        คัดลอกโค้ด SQL ด้านล่างนี้ไปวาง แล้วกดปุ่ม Run
                    

                    

                        
                        
CREATE OR REPLACE FUNCTION exec_sql(query text)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
AS $$
DECLARE
    result jsonb;
BEGIN
    EXECUTE 'WITH q AS (' || query || ') SELECT jsonb_agg(q) FROM q' INTO result;
    RETURN coalesce(result, '[]'::jsonb);
EXCEPTION WHEN OTHERS THEN
    BEGIN
        EXECUTE query;
        RETURN jsonb_build_object('success', true, 'message', 'Command executed successfully');
    EXCEPTION WHEN OTHERS THEN
        RETURN jsonb_build_object('error', true, 'message', SQLERRM);
    END;
END;
$$;

GRANT EXECUTE ON FUNCTION exec_sql(text) TO anon, authenticated, service_role;
